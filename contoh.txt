        function showMenuCategory(categoryId) {
            const menu = mainMenus.find(m => m.id === categoryId);
            if (!menu) return;
            
            addUserMessage(`${menu.icon} ${menu.label}`);
            
            let buttonsHTML = `Pilih informasi yang Anda butuhkan:<br><br><div class="menu-buttons">`;
            menu.keys.forEach(key => {
                const item = knowledgeBase[key];
                if (item) {
                    buttonsHTML += `<button class="menu-button" onclick="showContent('${key}')">
                        ${item.title}
                    </button>`;
                }
            });
            buttonsHTML += '</div>';
            
            addBotMessage(buttonsHTML, true);
        }

        function showContent(contentKey, fromSearch = false) {
            const content = knowledgeBase[contentKey];
            if (!content) return;
            
            if (!fromSearch) {
                addUserMessage(content.title);
            }
            
            // Show summary first
            addBotMessage(content.summary);
            
            // Show detail button
            const detailBtn = `<button class="detail-button" onclick="showDetailContent('${contentKey}')">
                📖 Lihat Detail Lengkap
            </button>`;
            addBotMessage(detailBtn, true);
            
            // Show related quick replies
            showRelatedQuestions(contentKey);
        }

        function showDetailContent(contentKey) {
            const content = knowledgeBase[contentKey];
            if (!content) return;
            
            addBotMessage(content.content);
            
            const backButtons = `
                <div class="menu-buttons">
                    <button class="menu-button" onclick="showMainMenu()">🏠 Menu Utama</button>
                </div>
            `;
            
            setTimeout(() => {
                addBotMessage(backButtons, true);
            }, 300);
        }

        // Process user input
        function sendMessage() {
            const input = document.getElementById('userInput');
            const message = input.value.trim();
            
            if (message === '') return;
            
            addUserMessage(message);
            input.value = '';
            
            // Show typing indicator
            showTypingIndicator();
            
            // Process after delay
            setTimeout(() => {
                removeTypingIndicator();
                processUserMessage(message);
            }, 800);
        }

        function sendQuickReply(message) {
            addUserMessage(message);
            
            showTypingIndicator();
            setTimeout(() => {
                removeTypingIndicator();
                processUserMessage(message);
            }, 800);
        }

        function processUserMessage(message) {
            const normalized = message.toLowerCase().trim();
            const matches = findMatches(normalized);
            
            if (matches.length === 0) {
                // No match found
                showNotFoundResponse(message);
            } else if (matches.length === 1) {
                // Single match - show directly
                showContent(matches[0], true);
            } else {
                // Multiple matches - show options
                showMultipleOptions(matches);
            }
        }

        function findMatches(input) {
            let matches = [];
            
            // Check exact match
            if (keywordMapping[input]) {
                matches = keywordMapping[input];
            } else {
                // Check partial match
                for (let keyword in keywordMapping) {
                    if (input.includes(keyword) || keyword.includes(input)) {
                        matches.push(...keywordMapping[keyword]);
                    }
                }
            }
            
            // Remove duplicates
            return [...new Set(matches)];
        }

        function showMultipleOptions(matches) {
            let html = '🤖 Saya menemukan beberapa topik yang mungkin Anda maksud:<br><br><div class="menu-buttons">';
            
            matches.forEach(key => {
                const item = knowledgeBase[key];
                if (item) {
                    html += `<button class="menu-button" onclick="showContent('${key}', true)">
                        ${item.title}
                    </button>`;
                }
            });
            
            html += '</div>';
            addBotMessage(html, true);
        }
